with open('test/shaping/data/text-rendering-tests/fonts/TestMORXThirtyfour.ttf','rb') as f:

# Focus on how the insertion subtable state machine is set up
morx_offset = 2416
morx_data = data[morx_offset:]

# Parse entry 1 flags
# flags=0x240 => 0x0200 = CurrentInsertCount bit 9 which is >> 5 = 1 glyph
# Let's decode flags fields
# CurrentInsertCount = 0x3E0 (5-bit field starting at bit 5)
# so (0x240 & 0x3E0) >> 5 = (0x200) >> 5 = 16 glyphs? wait let me check
flags = 0x240
CurrentInsertCount = 0x3E0
CurrentInsertBefore = 0x0800
MarkedInsertBefore = 0x0400
DontAdvance = 0x4000
SetMark = 0x8000
count = (flags & CurrentInsertCount) >> 5
before = bool(flags & CurrentInsertBefore)
print(f'Entry 1 flags = {hex(flags)}')
print(f'  count={count} before={before}')
print(f'  DontAdvance={bool(flags & DontAdvance)}')
print(f'  SetMark={bool(flags & SetMark)}')
print()
# State machine state 0 entry for class 4 (h=0x68 mapped) -> Entry 1
# Entry 1: newState=0, flags=0x240, currentInsertIndex=0, markedInsertIndex=0xffff
# That means: at end-of-text (class 0 = CLASS_END_OF_TEXT), state 0 entry 0 -> Entry 0 (no action)
# At class 4 (h glyph), state 0 -> Entry 1 (currentInsert at index 0, 18 glyphs)

# Entry 1 happens when class 4 glyph is seen from state 0.
# after it, newState=0, DontAdvance=False, so buffer.next_glyph() is called normally.
# So h glyph gets insertions.
# The regression we want: at end-of-text, currentInsert is triggered and copy_glyph() is called with idx==len.
# So we need to arrange so CLASS_END_OF_TEXT triggers an entry with currentInsertIndex != 0xFFFF
# and !(flags & CurrentInsertBefore) so copy_glyph() is called.

# The state transition at end-of-text uses klass=0 (CLASS_END_OF_TEXT=0)
